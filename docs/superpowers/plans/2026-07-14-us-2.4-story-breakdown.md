# US-2.4 — Story breakdown from an approved PRD — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let a PM split a `ready` feature's approved PRD into LLM-proposed, hand-edited child `story` issues, see their roll-up progress on the feature, add more stories later, and have the feature become a non-dispatchable container that auto-moves to `done` when every non-abandoned child is `merged`.

**Architecture:** Two Postgres triggers on `issues` (migration `034`) carry the two rules that must hold no matter which client writes — `guard_feature_dispatch` blocks a feature with children from entering `queued`, and `rollup_feature_done` flips the parent to `done` (plus an event) once every non-abandoned child is `merged`. The LLM split is the only orchestration, so it is one FastAPI endpoint (`POST /api/v1/llm/issues/{id}/breakdown`) that reads the issue + its approved PRD artifact through PostgREST under the caller's JWT and calls litellm via the existing `app/llm.py` plumbing; it *proposes only* and creates nothing. Accepting the proposal, adding a story by hand, and rendering children are plain CRUD from `apps/web` through the Supabase JS SDK under RLS.

**Tech Stack:** Postgres/PL-pgSQL (Supabase project `wdudmfhhqxrqzoyhuzwx`); FastAPI + litellm + httpx PostgREST client (`app/supabase.py`); pytest (monkeypatched, no live DB); Next.js 16 App Router + Supabase JS SDK + Base UI (`@base-ui/react`) shadcn components + `react-markdown`.

```sql
-- 034_story_breakdown: the two rules that make a broken-down feature a
-- container rather than a unit of work (US-2.4).
--
-- Both are triggers, not application code, because both must hold no
-- matter which client writes: the web app under RLS, dispatch_issue, the
-- GitHub sync, or a hand-run SQL statement.
--
--   1. guard_feature_dispatch — once a feature has child stories it is a
--      container; it can never be dispatched. The message contains "not
--      dispatchable" so routers/issues.py maps the RpcError to 409 with
--      no code change (same shape dispatch_issue already raises).
--   2. rollup_feature_done — a feature moves to 'done' by itself when
--      every one of its non-abandoned children reaches 'merged'.

create or replace function public.guard_feature_dispatch()
returns trigger
language plpgsql
as $$
begin
  if new.status <> 'queued' or old.status = 'queued' then
    return new;
  end if;
  if new.type <> 'feature' then
    return new;
  end if;

  if exists (
    select 1
    from public.issues c
    where c.parent_id = new.id
      and c.abandoned_at is null
  ) then
    raise exception
      'issue is not dispatchable: a feature with child stories is a container — dispatch its stories instead';
  end if;

  return new;
end;
$$;

create trigger issues_guard_feature_dispatch
  before update of status on public.issues
  for each row execute function public.guard_feature_dispatch();

create or replace function public.rollup_feature_done()
returns trigger
language plpgsql
as $$
declare
  v_parent public.issues%rowtype;
  v_open integer;
  v_active integer;
begin
  -- Only a child story can complete a feature.
  if new.parent_id is null then
    return new;
  end if;

  -- Nothing we care about changed.
  if new.status is not distinct from old.status
     and new.abandoned_at is not distinct from old.abandoned_at then
    return new;
  end if;

  select * into v_parent
  from public.issues
  where id = new.parent_id
  for update;

  if not found or v_parent.status = 'done' then
    return new;
  end if;

  select
    count(*) filter (where c.abandoned_at is null),
    count(*) filter (where c.abandoned_at is null and c.status <> 'merged')
  into v_active, v_open
  from public.issues c
  where c.parent_id = new.parent_id;

  -- An all-abandoned feature never shipped anything: it is not 'done'.
  if v_active = 0 or v_open > 0 then
    return new;
  end if;

  update public.issues
  set status = 'done'
  where id = v_parent.id;

  insert into public.issue_events (org_id, issue_id, type, payload)
  values (
    v_parent.org_id,
    v_parent.id,
    'feature-done',
    jsonb_build_object(
      'story_count', v_active,
      'completed_by_issue_id', new.id
    )
  );

  return new;
end;
$$;

-- AFTER, so the child's own row is already committed to the snapshot the
-- count above reads. The parent update re-fires this trigger for the
-- parent row, which returns early on parent_id is null.
create trigger issues_rollup_feature_done
  after update of status, abandoned_at on public.issues
  for each row execute function public.rollup_feature_done();
```

```sql
  values (v_org, v_project, 'story', 'Child A', 'draft', v_feature)
  returning id into v_a;

  insert into public.issues (org_id, project_id, type, title, status, parent_id)
  values (v_org, v_project, 'story', 'Child B', 'draft', v_feature)
  returning id into v_b;

  -- 1. container rule
  begin
    update public.issues set status = 'queued' where id = v_feature;
  exception when others then
    v_blocked := true;
  end;
  if not v_blocked then
    raise exception 'FAIL: a feature with children was allowed into queued';
  end if;

  -- 2. no rollup while a child is unmerged
  update public.issues set status = 'merged' where id = v_a;
  select status into v_status from public.issues where id = v_feature;
  if v_status <> 'ready' then
    raise exception 'FAIL: feature rolled up to "%" with an unmerged child', v_status;
  end if;

  -- 3. rollup when the last non-abandoned child merges
  update public.issues set status = 'merged' where id = v_b;
  select status into v_status from public.issues where id = v_feature;
  if v_status <> 'done' then
    raise exception 'FAIL: feature did not roll up to done, got "%"', v_status;
  end if;

  select count(*) into v_events
  from public.issue_events
  where issue_id = v_feature and type = 'feature-done';
  if v_events <> 1 then
    raise exception 'FAIL: expected 1 feature-done event, got %', v_events;
  end if;

  -- 4. a story with no parent is untouched by the guard
  update public.issues set status = 'queued' where id = v_a;

  raise exception 'ROLLBACK_OK: guard + rollup assertions all passed';
end $$;
```

```python
async def propose_story_breakdown(
    settings: Settings,
    user_token: str,
    title: str,
    body: str,
    prd: str,
) -> list[dict[str, Any]]:
    """Propose a split of an approved PRD into story drafts (US-2.4).

    Proposes only — nothing is written. The PM edits the drafts and the
    web app creates the child issues through the Supabase SDK under RLS.
    """
    config = await get_llm_config(settings, user_token)

    provider = config["provider_type"]
    prefix = _LITELLM_PREFIX.get(provider)
    if not prefix:
        raise LlmCallError(f'unknown provider type "{provider}"')

    api_key = None
    if config.get("vault_secret_id"):
        api_key = read_vault_secret(settings, config["vault_secret_id"])
    if not api_key and provider != "ollama":
        raise LlmNotConfigured()

    prompt = _BREAKDOWN_PROMPT.format(
        title=title.strip(),
        body=(body or "").strip() or "(none)",
        prd=prd.strip(),
    )

    kwargs: dict[str, Any] = {
        "model": f"{prefix}/{config['model']}",
        "messages": [{"role": "user", "content": prompt}],
        "temperature": 0.2,
        "timeout": 120,
    }
    if api_key:
        kwargs["api_key"] = api_key
    if config.get("base_url"):
        kwargs["api_base"] = config["base_url"]

    try:
        resp = await litellm.acompletion(**kwargs)
        text = resp.choices[0].message.content or ""
        parsed = _parse_json_reply(text)
    except (LlmCallError, json.JSONDecodeError):
        raise
    except Exception as e:  # litellm raises provider-specific exceptions
        raise LlmCallError(str(e))

    raw = parsed.get("stories")
    if not isinstance(raw, list):
        raise LlmCallError("provider reply had no stories")

    stories: list[dict[str, Any]] = []
    for item in raw:
        if not isinstance(item, dict):
            continue
        item_title = str(item.get("title", "")).strip()
        if not item_title:
            continue
        criteria = item.get("acceptance_criteria")
        if not isinstance(criteria, list):
            criteria = []
        stories.append(
            {
                "title": item_title,
                "story": str(item.get("story", "")).strip(),
                "acceptance_criteria": [
                    str(c).strip() for c in criteria if str(c).strip()
                ],
            }
        )

    if not stories:
        raise LlmCallError("provider reply had no stories")
    return stories
```

```typescript
import { createClient } from "@/lib/supabase/client";

/** One proposed/edited slice of a feature's PRD, before it becomes an
 * issue row. Matches the LLM breakdown endpoint's response shape. */
export type StoryDraft = {
  title: string;
  story: string;
  acceptance_criteria: string[];
};

export function emptyStoryDraft(): StoryDraft {
  return { title: "", story: "", acceptance_criteria: [""] };
}

/** Returns the first problem with the drafts, or null when they're valid. */
export function validateStoryDrafts(drafts: StoryDraft[]): string | null {
  if (!drafts.length) return "Add at least one story before accepting.";
  for (let i = 0; i < drafts.length; i++) {
    if (!drafts[i].title.trim()) return `Story ${i + 1} needs a title.`;
    if (!drafts[i].acceptance_criteria.some((c) => c.trim())) {
      return `Story ${i + 1} needs at least one acceptance criterion.`;
    }
  }
  return null;
}

export type CreateStoryIssuesArgs = {
  orgId: string;
  projectId: string;
  featureId: string;
  epicId: string | null;
  drafts: StoryDraft[];
  source: "breakdown" | "manual";
};

/**
 * Creates child story issues under a feature and writes the events on both
 * sides (US-2.4): `created` on each story, one `stories-created` on the
 * feature carrying the new ids. Plain CRUD under RLS — the `parent_id`
 * trigger from migration 031 is what enforces that only a story may hang
 * off a feature. Throws on failure; returns the created ids.
 */
export async function createStoryIssues({
  orgId,
  projectId,
  featureId,
  epicId,
  drafts,
  source,
}: CreateStoryIssuesArgs): Promise<string[]> {
  const supabase = createClient();

  const rows = drafts.map((d) => ({
    org_id: orgId,
    project_id: projectId,
    type: "story",
    parent_id: featureId,
    epic_id: epicId,
    title: d.title.trim(),
    body: d.story.trim() || null,
    acceptance_criteria: d.acceptance_criteria
      .map((c) => c.trim())
      .filter(Boolean),
  }));

  const { data, error } = await supabase
    .from("issues")
    .insert(rows)
    .select("id, title");
  if (error) throw new Error(error.message);
  const created = data ?? [];

  const { error: eventError } = await supabase.from("issue_events").insert([
    ...created.map((r) => ({
      org_id: orgId,
      issue_id: r.id,
      type: "created",
      payload: { title: r.title, parent_id: featureId, source },
    })),
    {
      org_id: orgId,
      issue_id: featureId,
      type: "stories-created",
      payload: {
        story_ids: created.map((r) => r.id),
        count: created.length,
        source,
      },
    },
  ]);
  if (eventError) throw new Error(eventError.message);

  return created.map((r) => r.id);
}
```

```typescript
"use client";

import { useState } from "react";
import { useRouter } from "next/navigation";
import { Loader2, Plus, Split, Trash2 } from "lucide-react";
import { apiFetch } from "@/lib/api";
import { Button } from "@/components/ui/button";
import {
  createStoryIssues,
  emptyStoryDraft,
  validateStoryDrafts,
  type StoryDraft,
} from "@/lib/story-drafts";
import { StoryDraftFields } from "./story-draft-fields";

export function StoryBreakdownPanel({
  orgId,
  projectId,
  featureId,
  epicId,
  hasApprovedPrd,
}: {
  orgId: string;
  projectId: string;
  featureId: string;
  epicId: string | null;
  hasApprovedPrd: boolean;
}) {
  const router = useRouter();
  const [drafts, setDrafts] = useState<StoryDraft[] | null>(null);
  const [proposing, setProposing] = useState(false);
  const [accepting, setAccepting] = useState(false);
  const [error, setError] = useState<string | null>(null);

  async function handlePropose() {
    setError(null);
    setProposing(true);
    try {
      const body = await apiFetch(`/api/v1/llm/issues/${featureId}/breakdown`, {
        method: "POST",
      });
      setDrafts(body.stories as StoryDraft[]);
    } catch (e) {
      setError((e as Error).message);
    } finally {
      setProposing(false);
    }
  }

  async function handleAccept() {
    if (!drafts) return;
    const problem = validateStoryDrafts(drafts);
    if (problem) {
      setError(problem);
      return;
    }
    setError(null);
    setAccepting(true);
    try {
      await createStoryIssues({
        orgId,
        projectId,
        featureId,
        epicId,
        drafts,
        source: "breakdown",
      });
      setDrafts(null);
      router.refresh();
    } catch (e) {
      setError((e as Error).message);
    } finally {
      setAccepting(false);
    }
  }

    setDrafts((prev) => (prev ?? []).map((d, idx) => (idx === i ? next : d)));

  if (!drafts) {
    return (
      <div className="flex flex-col items-start gap-2">
        <Button
          variant="outline"
          size="sm"
          disabled={!hasApprovedPrd || proposing}

        )}
      </div>
    );
  }

  return (
    <div className="flex flex-col gap-4 rounded-lg border border-dashed p-4">
      <div className="flex items-center justify-between gap-3">
        <div>
          <p className="text-sm font-medium">
            Proposed split — {drafts.length}{" "}
            {drafts.length === 1 ? "story" : "stories"}

          <p className="text-xs text-muted-foreground">

            setError(null);

          Discard
        </Button>
```
