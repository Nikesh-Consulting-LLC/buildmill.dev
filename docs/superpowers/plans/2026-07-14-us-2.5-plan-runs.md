# US-2.5 — Two-phase dispatch: plan runs — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn today's one-shot dispatch into two phases over the same runner/provider machinery — a `plan` run that produces an agent-written implementation plan + test plan the operator approves, then a `code` run that builds to the approved plan.

**Architecture:** `dispatch_issue` (the single transactional PL/pgSQL entry point, recreated by us-2.1 in migration 031) becomes phase-aware: with no approved `plan` artifact it creates a `kind = 'plan'` run and moves the issue to `planning`; with one it creates a `kind = 'code'` run carrying the approved plan and test plan, exactly as today. Three new PL/pgSQL functions (`approve_plan`, `send_back_plan`, `replan_issue`) drive the plan gate, writing `approvals` / `artifacts` / `issue_events`; the runner callback path stays in Python (`apps/api/app/db.py`), where `complete_run` branches on `runs.kind` to store the two artifacts and move the issue to `plan-review`. The web app gains a plan-review panel in the existing review inbox and a Re-plan action on issue detail.

**Tech Stack:** Postgres / PL/pgSQL (Supabase project `Software-Factory`, `wdudmfhhqxrqzoyhuzwx`); FastAPI + httpx PostgREST client (`apps/api/app/supabase.py`) and psycopg3 direct-Postgres (`apps/api/app/db.py`); pytest 9 (`apps/api/tests`, plus a new `apps/runner/tests`); Next.js 16 App Router + Supabase JS SDK + Base UI (`@base-ui/react`) + `react-markdown`.

```sql
-- 035_plan_runs: two-phase dispatch — a plan run before any code run (US-2.5).
--
-- dispatch_issue v2 (v1 is 031_issues.sql) branches on whether the issue has
-- an approved 'plan' artifact:
--   no approved plan -> kind = 'plan' run; the issue moves to 'planning'
--   approved plan    -> kind = 'code' run carrying the approved plan and
--                       test plan; the issue moves to 'queued' (today's flow)
-- The return type changes from uuid to jsonb ({run_id, kind}) so the caller
-- knows which phase it started. Postgres cannot change a return type with
-- `create or replace`, hence the explicit drop.
--
-- SECURITY INVOKER, as before: authorization is RLS — a non-member cannot
-- see the issue at all ("issue not found").

drop function if exists public.dispatch_issue(uuid);

create or replace function public.dispatch_issue(p_issue uuid)
returns jsonb
language plpgsql
as $$
declare
  v_issue public.issues%rowtype;
  v_project public.projects%rowtype;
  v_prev public.runs%rowtype;
  v_feedback text;
  v_guidelines text;
  v_learnings text;
  v_prd text;
  v_plan text;
  v_test_plan text;
  v_prev_plan text;
  v_prev_test_plan text;
  v_has_plan boolean;
  v_kind text;
  v_context jsonb;
  v_run uuid;
begin
  select * into v_issue from public.issues where id = p_issue for update;
  if not found then
    raise exception 'issue not found';
  end if;

  v_has_plan := exists (
    select 1 from public.artifacts
    where issue_id = p_issue and kind = 'plan' and status = 'approved'
  );

  -- The explicit refusal the story asks for: a 'planned' issue whose plan
  -- was superseded (or never approved) must go back through the gate.
  if v_issue.status = 'planned' and not v_has_plan then
    raise exception 'issue has no approved plan — dispatch a plan run first';
  end if;

  if v_has_plan then
    v_kind := 'code';
    if v_issue.status not in ('planned', 'needs-fixes', 'failed') then
      raise exception 'issue is not dispatchable from status "%"', v_issue.status;
    end if;
  else
    v_kind := 'plan';
    if v_issue.status not in ('draft', 'ready', 'failed') then
      raise exception 'issue is not dispatchable from status "%"', v_issue.status;
    end if;
  end if;

  select * into v_project from public.projects where id = v_issue.project_id;

  v_guidelines := public.assemble_project_guidelines(v_issue.project_id);
  v_learnings := public.assemble_project_learnings(v_issue.project_id);

  -- Feature-descended work carries the approved PRD (us-2.3/us-2.4): a
  -- child story reads its parent feature's, a childless feature its own.
  select a.content into v_prd
  from public.artifacts a
  where a.issue_id = coalesce(v_issue.parent_id, v_issue.id)
    and a.kind = 'prd'
    and a.status = 'approved'
  order by a.version desc
  limit 1;

  v_context := jsonb_build_object(
    'kind', v_kind,
    'issue_type', v_issue.type,

    -- never re-plans.

    from public.runs
```

```sql
-- 036_plan_gate: approve / send back / re-plan the plan gate (US-2.5).
--
-- The human decision points around a plan run. All three are SECURITY
-- INVOKER like dispatch_issue: RLS is the authorization.

-- Approve: both draft artifacts become 'approved', one approvals row
-- records the decision (subject = the implementation plan), the issue moves
-- to 'planned' and becomes dispatchable for code.
create or replace function public.approve_plan(p_issue uuid)
returns void
language plpgsql
as $$
declare
  v_issue public.issues%rowtype;
  v_plan public.artifacts%rowtype;
  v_test public.artifacts%rowtype;
begin
  select * into v_issue from public.issues where id = p_issue for update;
  if not found then
    raise exception 'issue not found';
  end if;
  if v_issue.status <> 'plan-review' then
    raise exception 'issue is not in plan review (status "%")', v_issue.status;
  end if;

  select * into v_plan
  from public.artifacts
  where issue_id = p_issue and kind = 'plan' and status = 'draft'
  order by version desc limit 1;

  select * into v_test
  from public.artifacts
  where issue_id = p_issue and kind = 'test_plan' and status = 'draft'
  order by version desc limit 1;

  if v_plan.id is null or v_test.id is null then
    raise exception 'no draft plan to approve';
  end if;

  update public.artifacts set status = 'approved'
  where id in (v_plan.id, v_test.id);

  insert into public.approvals
    (org_id, issue_id, gate, subject_type, subject_id, decision, actor)
  values
    (v_issue.org_id, p_issue, 'plan', 'artifact', v_plan.id, 'approved', auth.uid());

  update public.issues set status = 'planned' where id = p_issue;

  insert into public.issue_events (org_id, issue_id, type, payload)
  values (v_issue.org_id, p_issue, 'plan-approved',
          jsonb_build_object('plan_id', v_plan.id, 'test_plan_id', v_test.id));
end;
$$;

-- Send back: the comment is required — it becomes the next plan run's
-- feedback (us-1.13's informed retry, one gate up). The issue returns to
-- exactly where dispatch found it, so the same Dispatch action re-plans.
-- The artifacts stay 'draft' on purpose: dispatch_issue reads them as
-- 'previous_plan', and db.complete_run supersedes them when the new
-- version lands.
create or replace function public.send_back_plan(p_issue uuid, p_comment text)
returns void
language plpgsql
as $$
declare
  v_issue public.issues%rowtype;
  v_plan public.artifacts%rowtype;
  v_from text;
begin
  if p_comment is null or length(trim(p_comment)) = 0 then
    raise exception 'a comment is required to send a plan back';
  end if;

  select * into v_issue from public.issues where id = p_issue for update;
  if not found then
    raise exception 'issue not found';
  end if;
  if v_issue.status <> 'plan-review' then
    raise exception 'issue is not in plan review (status "%")', v_issue.status;
  end if;

  select * into v_plan
  from public.artifacts
  where issue_id = p_issue and kind = 'plan' and status = 'draft'
  order by version desc limit 1;

  if v_plan.id is null then
    raise exception 'no draft plan to send back';
  end if;

  insert into public.approvals
    (org_id, issue_id, gate, subject_type, subject_id, decision, comment, actor)
  values
    (v_issue.org_id, p_issue, 'plan', 'artifact', v_plan.id, 'sent-back',
     trim(p_comment), auth.uid());

  select e.payload->>'from_status' into v_from
  from public.issue_events e
  where e.issue_id = p_issue and e.type = 'plan-dispatched'
  order by e.created_at desc
  limit 1;

  update public.issues set status = coalesce(v_from, 'draft') where id = p_issue;

  insert into public.issue_events (org_id, issue_id, type, payload)
  values (v_issue.org_id, p_issue, 'plan-sent-back',
          jsonb_build_object('plan_id', v_plan.id, 'comment', trim(p_comment)));
end;
$$;

-- Re-plan: an explicit restart of the plan phase on a non-active issue.
-- Supersedes the prior artifacts (so dispatch_issue sees no approved plan,
-- forcing re-approval before the next code run) and dispatches a fresh
-- plan run in the same transaction.
create or replace function public.replan_issue(p_issue uuid)
returns jsonb
language plpgsql
as $$
declare
  v_issue public.issues%rowtype;
  v_base text;
begin
  select * into v_issue from public.issues where id = p_issue for update;
  if not found then
    raise exception 'issue not found';
  end if;
  if v_issue.status in ('planning', 'queued', 'running') then
    raise exception 'issue is active (status "%") — cannot re-plan', v_issue.status;
  end if;

  update public.artifacts set status = 'superseded'
  where issue_id = p_issue
    and kind in ('plan', 'test_plan')
    and status in ('draft', 'approved');

  v_base := case when v_issue.type = 'feature' then 'ready' else 'draft' end;
  update public.issues set status = v_base where id = p_issue;

  insert into public.issue_events (org_id, issue_id, type, payload)
  values (v_issue.org_id, p_issue, 're-planned',
          jsonb_build_object('from_status', v_issue.status));

  return public.dispatch_issue(p_issue);
end;
$$;

revoke execute on function public.approve_plan(uuid) from public, anon;
grant execute on function public.approve_plan(uuid) to authenticated;
revoke execute on function public.send_back_plan(uuid, text) from public, anon;
grant execute on function public.send_back_plan(uuid, text) to authenticated;
revoke execute on function public.replan_issue(uuid) from public, anon;
grant execute on function public.replan_issue(uuid) to authenticated;
```

```python
    stdout: str | None,

) -> bool:
    """Record a runner callback. Returns False if the run isn't running.

    A 'code' run behaves exactly as it did in Phase 1. A 'plan' run stores
    the two artifacts it produced and puts the issue in front of the plan
    gate instead of the diff review (US-2.5)."""
    with _connect(settings) as conn:
        kind_row = conn.execute(
            "select kind from public.runs where id = %s", (run_id,)
```

```python
@router.post("/{issue_id}/dispatch", status_code=202)
async def dispatch(
    issue_id: UUID,
    user: AuthUser = Depends(verify_token),
    settings: Settings = Depends(get_settings),
):
    """US-2.5: one action, two phases. dispatch_issue decides which — a plan
    run when the issue has no approved plan, a code run when it has."""
    prior = await postgrest_get(
        settings, user.token, "issues",
        {"select": "status", "id": f"eq.{issue_id}", "limit": "1"},
    )
    prior_status = prior[0]["status"] if prior else None

    try:
        result = await rpc(
            settings, user.token, "dispatch_issue", {"p_issue": str(issue_id)}
        )
    except RpcError as e:
        if "not found" in e.message:
            raise HTTPException(status_code=404, detail="Issue not found")
        if "not dispatchable" in e.message or "no approved plan" in e.message:
            raise HTTPException(status_code=409, detail=e.message)
        raise HTTPException(status_code=400, detail=e.message)

    if prior_status == "failed":
        # Best-effort: a redispatch out of "failed" reopens the synced
        # issue that was closed when it first failed (US-1.20).
        await issue_sync.push_issue_state_via_postgrest(
            settings, user.token, str(issue_id), "open"
        )

    return {
        "run_id": result["run_id"],
        "kind": result["kind"],
        "status": "planning" if result["kind"] == "plan" else "queued",
    }
```

```tsx
import { useState } from "react";
import { useRouter } from "next/navigation";
import { Loader2, Rocket, Route } from "lucide-react";
import { apiFetch } from "@/lib/api";
import { Button } from "@/components/ui/button";

// US-2.5: one action, two phases. The server decides which — these sets
// only decide whether the button is live and what it says.
const PLAN_PHASE = new Set(["draft", "ready", "failed"]);
const CODE_PHASE = new Set(["planned", "needs-fixes"]);

export function DispatchButton({
  issueId,
  status,
}: {
  issueId: string;
  status: string;
}) {
  const router = useRouter();
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const plans = PLAN_PHASE.has(status);
  const dispatchable = plans || CODE_PHASE.has(status);

  async function handleDispatch() {
    setError(null);
    setBusy(true);
    try {
      await apiFetch(`/api/v1/issues/${issueId}/dispatch`, { method: "POST" });
      router.refresh();
    } catch (e) {
      setError((e as Error).message);
    } finally {
      setBusy(false);
    }
  }

  return (
    <div className="flex flex-col items-end gap-1">
      <Button
        onClick={handleDispatch}
        disabled={!dispatchable || busy}
        title={
          !dispatchable
            ? `An issue in "${status}" cannot be dispatched`
            : plans
              ? "Ask the factory for an implementation plan and a test plan"
              : "Send the approved plan to the factory to build"
        }
      >
        {busy ? (
          <Loader2 className="size-4 animate-spin" />
        ) : plans ? (
          <Route className="size-4" />
        ) : (
          <Rocket className="size-4" />
        )}
        {plans ? "Plan" : "Dispatch"}
      </Button>
      {error && (
        <p className="max-w-64 text-right text-xs font-medium text-destructive">
          {error}
        </p>
      )}
    </div>
```

```tsx
import { useState } from "react";
import { useRouter } from "next/navigation";
import { Check, Loader2, Undo2 } from "lucide-react";
import { apiFetch } from "@/lib/api";
import { Button } from "@/components/ui/button";
import {
  Dialog,
  DialogContent,
  DialogDescription,
  DialogFooter,
  DialogHeader,
  DialogTitle,
  DialogTrigger,
} from "@/components/ui/dialog";
import { Label } from "@/components/ui/label";
import { Textarea } from "@/components/ui/textarea";

export function PlanReviewActions({ issueId }: { issueId: string }) {
  const router = useRouter();
  const [busy, setBusy] = useState<"approve" | "send-back" | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [approveOpen, setApproveOpen] = useState(false);
  const [sendBackOpen, setSendBackOpen] = useState(false);
  const [comment, setComment] = useState("");

  async function approve() {
    setError(null);
    setBusy("approve");
    try {
      await apiFetch(`/api/v1/issues/${issueId}/plan/approve`, {
        method: "POST",
      });
      router.push(`/issues/${issueId}`);
      router.refresh();
    } catch (e) {
      setError((e as Error).message);
      setBusy(null);
    }
  }

  async function sendBack() {
    if (!comment.trim()) {
      setError("A comment is required — it becomes the next plan's feedback.");
      return;
    }
    setError(null);
    setBusy("send-back");
    try {
      await apiFetch(`/api/v1/issues/${issueId}/plan/send-back`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ comment: comment.trim() }),
      });
      router.push(`/issues/${issueId}`);
      router.refresh();
    } catch (e) {
      setError((e as Error).message);
      setBusy(null);
    }
  }

  return (
    <div className="flex flex-col gap-2">
      <div className="flex items-center gap-2">
        <Dialog open={approveOpen} onOpenChange={setApproveOpen}>
          <DialogTrigger render={<Button />}>
            <Check className="size-4" />
            Approve plan
          </DialogTrigger>
          <DialogContent>
            <DialogHeader>
              <DialogTitle>Approve this plan?</DialogTitle>
              <DialogDescription>
                Records your approval and makes the issue dispatchable for
                code. The code run builds to exactly this plan.
              </DialogDescription>
            </DialogHeader>
            <DialogFooter>
              <Button
                variant="outline"
                onClick={() => setApproveOpen(false)}
                disabled={busy === "approve"}
              >
                Cancel
              </Button>
              <Button onClick={approve} disabled={busy === "approve"}>
                {busy === "approve" && (
                  <Loader2 className="size-4 animate-spin" />
                )}
                Approve plan
              </Button>
            </DialogFooter>
          </DialogContent>
        </Dialog>

        <Dialog open={sendBackOpen} onOpenChange={setSendBackOpen}>
          <DialogTrigger render={<Button variant="outline" />}>
            <Undo2 className="size-4" />
            Send back
          </DialogTrigger>
          <DialogContent>
            <DialogHeader>
              <DialogTitle>Send the plan back</DialogTitle>
              <DialogDescription>
                Your comment and this plan are attached to the next plan run,
                so the rewrite is informed, not blind.
              </DialogDescription>
            </DialogHeader>
            <div className="grid gap-2">
              <Label htmlFor="send-back-comment">
                What&apos;s wrong with the approach?
              </Label>
              <Textarea
                id="send-back-comment"
                rows={4}
                placeholder="The plan rewrites the exporter — extend the existing writer instead, and cover the rollback path."

                Send plan back

            </DialogFooter>
```
