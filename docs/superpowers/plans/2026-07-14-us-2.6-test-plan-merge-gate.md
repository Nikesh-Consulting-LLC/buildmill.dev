# US-2.6 — Test plan execution & soft merge gate — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn an approved `test_plan` artifact into runnable `test_cases`, surface those cases' latest results on the code review panel, and require a typed, logged reason (`approvals.gate = 'merge-override'`) when merging with failing or unrun linked tests.

**Architecture:** Materialization is a database concern — an `after update of status` trigger on `artifacts` fires `public.materialize_test_plan(artifact)`, which parses a fenced ` ```testcases ` JSON block out of the artifact's markdown and inserts `source = 'agent'` rows into `test_cases` stamped with the originating `artifact_id` (which also makes it idempotent). This deliberately avoids coupling to whatever code path [us-2.5](../../../stories/us-2.5-plan-runs.md) uses to approve a plan: anything that sets `status = 'approved'` materializes. The gate itself is split into a pure summarizer (`passed`/`failed`/`unrun` per issue) implemented twice — Python in `apps/api/app/test_state.py` for the authoritative server-side check on approve, TypeScript in `apps/web/src/lib/test-state.ts` for the review panel's strip — because the API must not trust a reason the client decided was necessary, and the panel must not round-trip to render.

**Tech Stack:** Postgres (plpgsql trigger + function, RLS via `public.is_org_member(org_id)`), FastAPI + pytest (monkeypatched, no live DB), Next.js 16 App Router + Supabase JS SDK, Base UI (shadcn) components.

```sql
-- 036_test_plan_materialization: an approved test_plan artifact becomes
-- runnable library cases (US-2.6).
--
-- The trigger keys off the artifact reaching status 'approved' rather than
-- off any particular approve code path, so every route into approval
-- (RPC, PostgREST patch, admin SQL) materializes exactly once. artifact_id
-- is what makes "exactly once" true: re-approving a superseded-then-restored
-- artifact finds its cases already there and does nothing.

alter table public.test_cases
  add column artifact_id uuid references public.artifacts(id) on delete set null;

create index test_cases_artifact_idx on public.test_cases (artifact_id);

create or replace function public.materialize_test_plan(p_artifact uuid)
returns integer
language plpgsql
security invoker
as $$
declare
  v_artifact record;
  v_project uuid;
  v_block text;
  v_json jsonb;
  v_case jsonb;
  v_count integer := 0;
begin
  select a.id, a.org_id, a.issue_id, a.kind, a.content
    into v_artifact
    from public.artifacts a
   where a.id = p_artifact;

  if not found then
    raise exception 'artifact % not found', p_artifact;
  end if;

  if v_artifact.kind <> 'test_plan' then
    raise exception 'artifact % is not a test_plan', p_artifact;
  end if;

  if exists (select 1 from public.test_cases where artifact_id = p_artifact) then
    return 0;
  end if;

  select i.project_id into v_project
    from public.issues i
   where i.id = v_artifact.issue_id;

  if not found then
    raise exception 'issue % not found for artifact %', v_artifact.issue_id, p_artifact;
  end if;

  v_block := substring(v_artifact.content from '```testcases[[:space:]]+([^`]*)```');

  if v_block is null then
    return 0;
  end if;

  begin
    v_json := v_block::jsonb;
  exception when others then
    raise exception 'test_plan artifact % has an unparsable testcases block', p_artifact;
  end;

  if jsonb_typeof(v_json) <> 'array' then
    raise exception 'test_plan artifact % testcases block is not a JSON array', p_artifact;
  end if;

  for v_case in select * from jsonb_array_elements(v_json)
  loop
    if coalesce(btrim(v_case->>'title'), '') = '' then
      continue;
    end if;

    insert into public.test_cases (
      org_id, project_id, issue_id, artifact_id, title, steps,
      expected_result, source, test_types, environments
    ) values (
      v_artifact.org_id,
      v_project,
      v_artifact.issue_id,
      p_artifact,
      btrim(v_case->>'title'),
      coalesce(v_case->>'steps', ''),
      coalesce(v_case->>'expected_result', ''),
      'agent',
      case when jsonb_typeof(v_case->'test_types') = 'array'
           then v_case->'test_types' else '[]'::jsonb end,
      case when jsonb_typeof(v_case->'environments') = 'array'
           then v_case->'environments' else '[]'::jsonb end
    );

    v_count := v_count + 1;
  end loop;

  return v_count;
end;
$$;

revoke execute on function public.materialize_test_plan(uuid) from public, anon;
grant execute on function public.materialize_test_plan(uuid) to authenticated;

create or replace function public.on_test_plan_approved()
returns trigger
language plpgsql
security invoker
as $$
begin
  if new.kind = 'test_plan'
     and new.status = 'approved'
     and old.status is distinct from 'approved' then
    perform public.materialize_test_plan(new.id);
  end if;
  return new;
end;
$$;

revoke execute on function public.on_test_plan_approved() from public, anon, authenticated;

create trigger artifacts_materialize_test_plan
  after update of status on public.artifacts
  for each row execute function public.on_test_plan_approved();
```

```python
"""Linked-test state for an issue — the data behind the soft merge gate (US-2.6).

The API recomputes this itself on approve rather than trusting the client's
claim that an override was needed: the reason is the client's, the verdict
is ours.
"""

from dataclasses import dataclass

from .config import Settings
from .supabase import postgrest_get

UNRUN = "unrun"

TEST_STATE_SELECT = "id,title,test_run_results(result,recorded_at,test_run_id)"


@dataclass(frozen=True)
class CaseState:
    id: str
    title: str
    result: str
    run_id: str | None


@dataclass(frozen=True)
class TestState:
    cases: tuple[CaseState, ...]
    passed: int
    failed: int
    unrun: int

    @property
    def needs_override(self) -> bool:
        return bool(self.cases) and (self.failed > 0 or self.unrun > 0)

    @property
    def summary(self) -> str:
        if not self.cases:
            return "no linked tests"
        parts = []
        if self.unrun:
            parts.append(f"{self.unrun} unrun")
        if self.failed:
            parts.append(f"{self.failed} failing")
        return ", ".join(parts) if parts else "all passing"


def _latest(results: list[dict]) -> tuple[str, str | None]:
    recorded = [
        r
        for r in results
        if r.get("result") in ("pass", "fail") and r.get("recorded_at")
    ]
    if not recorded:
        return UNRUN, None
    latest = max(recorded, key=lambda r: r["recorded_at"])
    return latest["result"], latest.get("test_run_id")


def summarize_test_state(rows: list[dict]) -> TestState:
    cases = []
    for row in rows:
        result, run_id = _latest(row.get("test_run_results") or [])
        cases.append(
            CaseState(
                id=str(row["id"]),
                title=row.get("title", ""),
                result=result,
                run_id=run_id,
            )
        )
    return TestState(
        cases=tuple(cases),
        passed=sum(1 for c in cases if c.result == "pass"),
        failed=sum(1 for c in cases if c.result == "fail"),
        unrun=sum(1 for c in cases if c.result == UNRUN),
    )


async def fetch_issue_test_state(
    settings: Settings, user_token: str, issue_id: str
) -> TestState:
    rows = await postgrest_get(
        settings,
        user_token,
        "test_cases",
        {
            "select": TEST_STATE_SELECT,
            "issue_id": f"eq.{issue_id}",
            "status": "eq.active",
            "order": "created_at.asc",
        },
    )
    return summarize_test_state(rows or [])
```

```python
        raise HTTPException(status_code=404, detail="Run not found")
    run = runs[0]

    # The verdict is recomputed here rather than taken from the client.
    state = await test_state.fetch_issue_test_state(
        settings, user.token, str(run["issue_id"])
    )

    if state.needs_override and not reason:
        raise HTTPException(
            status_code=409,
            detail=(
                f"Linked tests are not clean ({state.summary}) — "
                "a merge-override reason is required"
            ),
        )

    merge_result = await _merge_pr(settings, user.token, run["pr_url"])

    # US-1.48: record the squash-merge commit so deploy changelogs can map
    # commits back to the issue that shipped them.
    if merge_result.startswith("merged:"):

                {"merge_commit_sha": merge_sha},
            )
        except Exception:
            pass  # traceability is best-effort; the merge itself succeeded
```

```ts
export type CaseResult = "pass" | "fail" | "unrun";

export type TestRunResultRow = {
  result: string;
  recorded_at: string | null;
  test_run_id: string;
};

export type TestCaseWithResults = {
  id: string;
  title: string;
  test_run_results: TestRunResultRow[] | null;
};

export type IssueTestCaseState = {
  id: string;
  title: string;
  result: CaseResult;
  runId: string | null;
};

export type IssueTestState = {
  cases: IssueTestCaseState[];
  passed: number;
  failed: number;
  unrun: number;
  needsOverride: boolean;
  summary: string;
};

/** Mirrors TEST_STATE_SELECT in apps/api/app/test_state.py — the gate's
 * verdict must not depend on which side computed it. */
export const TEST_STATE_SELECT =
  "id, title, test_run_results(result, recorded_at, test_run_id)";

function latest(results: TestRunResultRow[]): {
  result: CaseResult;
  runId: string | null;
} {
  const recorded = results.filter(
    (r) => (r.result === "pass" || r.result === "fail") && r.recorded_at
  );
  if (!recorded.length) return { result: "unrun", runId: null };
  const newest = recorded.reduce((a, b) =>
    (a.recorded_at as string) >= (b.recorded_at as string) ? a : b
  );
  return { result: newest.result as CaseResult, runId: newest.test_run_id };
}

function summarize(total: number, failed: number, unrun: number): string {
  if (!total) return "no linked tests";
  const parts: string[] = [];
  if (unrun) parts.push(`${unrun} unrun`);
  if (failed) parts.push(`${failed} failing`);
  return parts.length ? parts.join(", ") : "all passing";
}

export function summarizeIssueTests(
  rows: TestCaseWithResults[]
): IssueTestState {
  const cases: IssueTestCaseState[] = rows.map((row) => {
    const { result, runId } = latest(row.test_run_results ?? []);
    return { id: row.id, title: row.title, result, runId };
  });
  const passed = cases.filter((c) => c.result === "pass").length;
  const failed = cases.filter((c) => c.result === "fail").length;
  const unrun = cases.filter((c) => c.result === "unrun").length;
  return {
    cases,
    passed,
    failed,
    unrun,
    needsOverride: cases.length > 0 && (failed > 0 || unrun > 0),
    summary: summarize(cases.length, failed, unrun),
  };
}
```

```tsx
            )}
            <DialogFooter>
              <Button
                variant="outline"
                onClick={() => setApproveOpen(false)}
                disabled={busy === "approve"}
              >
                Cancel
              </Button>
              <Button onClick={approve} disabled={busy === "approve"}>
                {busy === "approve" && (
                  <Loader2 className="size-4 animate-spin" />
                )}
                {needsOverride ? "Merge anyway" : "Approve & merge"}
              </Button>
            </DialogFooter>
          </DialogContent>
        </Dialog>

        <Dialog open={rejectOpen} onOpenChange={setRejectOpen}>
          <DialogTrigger render={<Button variant="outline" />}>
            <X className="size-4" />
            Reject
          </DialogTrigger>
          <DialogContent>
            <DialogHeader>
              <DialogTitle>Reject with feedback</DialogTitle>
              <DialogDescription>
                Your comment is attached to the retry so the next attempt is
                informed, not blind.
              </DialogDescription>
            </DialogHeader>
            <div className="grid gap-2">
              <Label htmlFor="reject-comment">What needs to change?</Label>
              <Textarea
                id="reject-comment"
                rows={4}
                placeholder="The version constant is hard-coded — read it from package metadata."
                value={comment}
                onChange={(e) => setComment(e.target.value)}
              />
            </div>
            <DialogFooter>
              <Button
                variant="outline"
                onClick={() => setRejectOpen(false)}
                disabled={busy === "reject"}
              >
                Cancel
              </Button>
              <Button
                variant="destructive"
                onClick={reject}
                disabled={busy === "reject"}
              >
                {busy === "reject" && (
                  <Loader2 className="size-4 animate-spin" />
                )}
                Reject issue
              </Button>
            </DialogFooter>
          </DialogContent>
        </Dialog>
      </div>
      {error && (
        <p className="text-sm font-medium text-destructive">{error}</p>
      )}
    </div>
  );
}
```
