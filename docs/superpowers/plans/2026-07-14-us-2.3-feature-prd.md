# US-2.3 — PRD on feature issues — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give feature issues a PRD lifecycle — the configured thinking-LLM drafts a four-section PRD artifact, the PM edits it per section, approves it (feature → `ready`) or sends it back with a comment that informs the next draft — and block any feature without an approved PRD from being dispatched.

**Architecture:** Migration `033` adds three SECURITY-INVOKER Postgres functions over us-2.1's `artifacts` / `approvals` / `issue_events` tables (`save_prd_draft`, `approve_prd`, `send_back_prd`) so each multi-table transition is one transaction, plus a `dispatch_issue` v2 that refuses a `feature` with no `approved` PRD artifact. FastAPI owns exactly one new endpoint — `POST /api/v1/issues/{issue_id}/prd/draft` — because it is the only step that calls the LLM (litellm via `app/llm.py`, Vault key read over the API's direct Postgres connection); it composes the LLM's JSON reply into markdown and persists it through `save_prd_draft`. Everything else is plain CRUD from `apps/web` under RLS: the PM's per-section edits `update` the draft artifact through the Supabase JS SDK, and approve / send-back call the RPCs via `supabase.rpc()` — no API endpoint needed for either.

**Tech Stack:** Postgres / PL-pgSQL (migration + RPCs, applied live via Supabase MCP `apply_migration`); FastAPI + litellm + httpx PostgREST client (`app/supabase.py`, `app/llm.py`); pytest with monkeypatched collaborators (no live DB); Next.js 16 App Router + Supabase JS SDK + Base UI (`@base-ui/react`) shadcn components + `react-markdown` (already a dependency).

```sql
-- 033_prd_artifacts: the PRD lifecycle on feature issues (US-2.3).
--
-- Three transitions, three functions, one transaction each — the same
-- shape as approve_run/reject_run (006_reviews.sql): SECURITY INVOKER so
-- RLS is the authorization, revoked from public/anon, granted to
-- authenticated. The LLM call that produces the markdown happens in the
-- API before save_prd_draft commits; this file never talks to a provider.
--
-- Versioning rule: every draft run is a new artifact version and
-- supersedes whatever came before it, so an approved PRD that gets
-- re-drafted stays readable at its old version while the feature returns
-- to prd-review for re-approval.

-- Idempotent policy set: correct whether 031 wrote one "for all" policy
-- or split policies. The PM edits a draft PRD directly through the JS SDK
-- ("build less API"), so UPDATE is scoped to status = 'draft' — an
-- approved or superseded artifact is immutable from the browser.
drop policy if exists "members manage their org artifacts" on public.artifacts;
drop policy if exists "members read their org artifacts" on public.artifacts;
drop policy if exists "members insert their org artifacts" on public.artifacts;
drop policy if exists "members update their org draft artifacts" on public.artifacts;

create policy "members read their org artifacts"
  on public.artifacts for select
  using (public.is_org_member(org_id));

create policy "members insert their org artifacts"
  on public.artifacts for insert
  with check (public.is_org_member(org_id));

create policy "members update their org draft artifacts"
  on public.artifacts for update
  using (public.is_org_member(org_id) and status = 'draft')
  with check (public.is_org_member(org_id) and status = 'draft');

-- No delete policy: artifact history is append-only from the browser.

create index if not exists artifacts_issue_kind_idx
  on public.artifacts (issue_id, kind, version desc);

create index if not exists approvals_subject_idx
  on public.approvals (subject_type, subject_id, created_at desc);

-- save_prd_draft: persist a freshly drafted PRD as the next version and
-- move the feature into prd-review. Supersedes every prior prd artifact
-- on the issue (draft or approved) — re-drafting always demands
-- re-approval.
create or replace function public.save_prd_draft(p_issue uuid, p_content text)
returns uuid
language plpgsql
as $$
declare
  v_issue public.issues%rowtype;
  v_version integer;
  v_artifact uuid;
begin
  if p_content is null or length(trim(p_content)) = 0 then
    raise exception 'prd content is required';
  end if;

  select * into v_issue from public.issues where id = p_issue for update;
  if not found then
    raise exception 'issue not found';
  end if;
  if v_issue.type <> 'feature' then
    raise exception 'only a feature issue has a prd (type "%")', v_issue.type;
  end if;
  if v_issue.status not in ('draft', 'prd-review', 'ready') then
    raise exception 'a prd cannot be drafted from status "%"', v_issue.status;
  end if;

  select coalesce(max(version), 0) + 1 into v_version
  from public.artifacts
  where issue_id = p_issue and kind = 'prd';

  update public.artifacts
  set status = 'superseded'
  where issue_id = p_issue and kind = 'prd' and status in ('draft', 'approved');

  insert into public.artifacts
    (org_id, issue_id, kind, content, version, status, created_by)
  values
    (v_issue.org_id, p_issue, 'prd', p_content, v_version, 'draft', 'llm')
  returning id into v_artifact;

  update public.issues set status = 'prd-review' where id = p_issue;

  insert into public.issue_events (org_id, issue_id, type, payload)
  values (v_issue.org_id, p_issue, 'prd-drafted',
          jsonb_build_object('artifact_id', v_artifact, 'version', v_version));

  return v_artifact;
end;
$$;

-- approve_prd: the gate. Records the approval, freezes the artifact, and
-- releases the feature to 'ready' — which is what US-2.4/US-2.5 will
-- require before breakdown or a plan run.
create or replace function public.approve_prd(p_artifact uuid)
returns void
language plpgsql
as $$
declare
  v_artifact public.artifacts%rowtype;
  v_issue public.issues%rowtype;
begin
  select * into v_artifact from public.artifacts where id = p_artifact for update;
  if not found then
    raise exception 'artifact not found';
  end if;
  if v_artifact.kind <> 'prd' then
    raise exception 'artifact is not a prd (kind "%")', v_artifact.kind;
  end if;
  if v_artifact.status <> 'draft' then
    raise exception 'prd is not awaiting approval (status "%")', v_artifact.status;
  end if;

  select * into v_issue from public.issues where id = v_artifact.issue_id for update;

  update public.artifacts set status = 'approved' where id = p_artifact;

  insert into public.approvals
    (org_id, issue_id, gate, subject_type, subject_id, decision, actor)
  values
    (v_artifact.org_id, v_artifact.issue_id, 'prd', 'artifact', p_artifact,
     'approved', auth.uid());

  update public.issues set status = 'ready' where id = v_artifact.issue_id;

  insert into public.issue_events (org_id, issue_id, type, payload)
  values (v_artifact.org_id, v_artifact.issue_id, 'prd-approved',
          jsonb_build_object('artifact_id', p_artifact,
                             'version', v_artifact.version));
end;
$$;

-- send_back_prd: comment required — it becomes the feedback the next
-- draft run receives (informed retry, mirroring reject_run/US-1.13). The
-- artifact stays 'draft' and the feature stays in prd-review.
create or replace function public.send_back_prd(p_artifact uuid, p_comment text)
returns void
language plpgsql
as $$
declare
  v_artifact public.artifacts%rowtype;
begin
  if p_comment is null or length(trim(p_comment)) = 0 then
    raise exception 'a comment is required to send a prd back';
  end if;

  select * into v_artifact from public.artifacts where id = p_artifact for update;
  if not found then
    raise exception 'artifact not found';
  end if;
  if v_artifact.kind <> 'prd' then
    raise exception 'artifact is not a prd (kind "%")', v_artifact.kind;
  end if;
  if v_artifact.status <> 'draft' then
    raise exception 'prd is not awaiting approval (status "%")', v_artifact.status;
  end if;

  insert into public.approvals
    (org_id, issue_id, gate, subject_type, subject_id, decision, comment, actor)
  values
    (v_artifact.org_id, v_artifact.issue_id, 'prd', 'artifact', p_artifact,
     'sent-back', trim(p_comment), auth.uid());

  update public.issues set status = 'prd-review' where id = v_artifact.issue_id;

  insert into public.issue_events (org_id, issue_id, type, payload)
  values (v_artifact.org_id, v_artifact.issue_id, 'prd-sent-back',
          jsonb_build_object('artifact_id', p_artifact,
                             'comment', trim(p_comment)));
end;
$$;

-- dispatch_issue v2: a feature without an approved PRD is not
-- dispatchable. Bugs, chores, and stories are untouched. Everything else
-- is 031's body, unchanged.
create or replace function public.dispatch_issue(p_issue uuid)
returns uuid
language plpgsql
as $$
declare
  v_issue public.issues%rowtype;
  v_project public.projects%rowtype;
  v_prev public.runs%rowtype;
  v_feedback text;
  v_guidelines text;
  v_learnings text;
  v_context jsonb;
  v_run uuid;
begin
  select * into v_issue from public.issues where id = p_issue for update;
  if not found then
    raise exception 'issue not found';
  end if;
  if v_issue.status not in ('draft', 'needs-fixes', 'failed') then
    raise exception 'issue is not dispatchable from status "%"', v_issue.status;
  end if;

  -- US-2.3 gate: a feature is an idea until its PRD is agreed.
  if v_issue.type = 'feature'
     and not exists (
       select 1 from public.artifacts
       where issue_id = p_issue and kind = 'prd' and status = 'approved'
     ) then
    raise exception 'feature has no approved prd — draft and approve one first';
  end if;

  select * into v_project from public.projects where id = v_issue.project_id;

  select * into v_prev
  from public.runs
  where issue_id = p_issue
  order by created_at desc
  limit 1;

  if v_prev.id is not null then
    select a.comment into v_feedback
    from public.approvals a
    where a.subject_type = 'run'
      and a.subject_id = v_prev.id
      and a.decision = 'rejected'
    order by a.created_at desc
    limit 1;
  end if;

  v_guidelines := public.assemble_project_guidelines(v_issue.project_id);
  v_learnings := public.assemble_project_learnings(v_issue.project_id);

  v_context := jsonb_build_object(
    'title', v_issue.title,
    'story', v_issue.body,
    'acceptance_criteria', v_issue.acceptance_criteria,
    'repo_full_name', v_project.repo_full_name,
    'default_branch', v_project.default_branch,
    'guidelines', v_guidelines,
    'learnings', v_learnings
  );

  if v_feedback is not null then
    v_context := v_context || jsonb_build_object(
      'feedback', v_feedback,
      'previous_branch', v_prev.branch_ref,
      'previous_pr_url', v_prev.pr_url
    );
  end if;

  insert into public.runs (org_id, issue_id, provider, status, input_context, kind)
  values (v_issue.org_id, p_issue, 'claude', 'queued', v_context, 'code')
  returning id into v_run;

  update public.issues set status = 'queued' where id = p_issue;

  insert into public.issue_events (org_id, issue_id, type, payload)
  values (v_issue.org_id, p_issue, 'dispatched', jsonb_build_object('run_id', v_run));

  return v_run;
end;
$$;

revoke execute on function public.save_prd_draft(uuid, text) from public, anon;
grant execute on function public.save_prd_draft(uuid, text) to authenticated;
revoke execute on function public.approve_prd(uuid) from public, anon;
grant execute on function public.approve_prd(uuid) to authenticated;
revoke execute on function public.send_back_prd(uuid, text) from public, anon;
grant execute on function public.send_back_prd(uuid, text) to authenticated;
revoke execute on function public.dispatch_issue(uuid) from public, anon;
grant execute on function public.dispatch_issue(uuid) to authenticated;
```

# US-2.3: the PRD's four sections, in document order. The API composes
# markdown from the provider's JSON so the web app never has to trust a
# provider's heading style — parse/serialize on the browser side keys off
# exactly these headings (apps/web/src/lib/prd.ts).
PRD_SECTIONS: tuple[tuple[str, str], ...] = (
    ("problem", "Problem"),
    ("goals", "Goals"),
    ("out_of_scope", "Out of scope"),
    ("acceptance_criteria", "Acceptance criteria"),
)

_DRAFT_PRD_PROMPT = """You are a product manager writing a PRD for one feature \
of a software product.

Expand the feature below into a Product Requirements Document.
Respond with ONLY a JSON object, no prose, no code fences:
{{"problem": "<the user or business problem, markdown prose>",
  "goals": "<markdown bullet list of what this feature must achieve>",
  "out_of_scope": "<markdown bullet list of what this deliberately does NOT cover>",
  "acceptance_criteria": "<markdown checklist, one `- [ ] ` line per criterion>"}}

Feature title:
{title}

Feature description:
{body}
{retry_block}"""

_DRAFT_PRD_RETRY_BLOCK = """
The previous draft of this PRD was sent back by the reviewer. Write a new \
draft that addresses the feedback — do not repeat the same mistakes.

Previous draft:
{previous_draft}

Reviewer feedback:
{feedback}"""


def compose_prd_markdown(parsed: dict[str, Any]) -> str:
    if not str(parsed.get("problem", "")).strip():
        raise LlmCallError("provider reply had no problem statement")
    blocks = [
        f"## {heading}\n\n{str(parsed.get(key, '')).strip()}"
        for key, heading in PRD_SECTIONS
    ]
    return "\n\n".join(blocks).strip()


async def draft_prd(
    settings: Settings,
    user_token: str,
    title: str,
    body: str,
    previous_draft: str | None = None,
    feedback: str | None = None,
) -> str:
    config = await get_llm_config(settings, user_token)

    provider = config["provider_type"]
    prefix = _LITELLM_PREFIX.get(provider)
    if not prefix:
        raise LlmCallError(f'unknown provider type "{provider}"')

    api_key = None
    if config.get("vault_secret_id"):
        api_key = read_vault_secret(settings, config["vault_secret_id"])
    if not api_key and provider != "ollama":
        raise LlmNotConfigured()

    retry_block = ""
    if feedback and feedback.strip():
        retry_block = _DRAFT_PRD_RETRY_BLOCK.format(
            previous_draft=(previous_draft or "(not kept)").strip(),
            feedback=feedback.strip(),
        )

    prompt = _DRAFT_PRD_PROMPT.format(
        title=title.strip(),
        body=(body or "").strip() or "(no description was written)",
        retry_block=retry_block,
    )

    kwargs: dict[str, Any] = {
        "model": f"{prefix}/{config['model']}",
        "messages": [{"role": "user", "content": prompt}],
        "temperature": 0.2,
        "timeout": 120,
    }
    if api_key:
        kwargs["api_key"] = api_key
    if config.get("base_url"):
        kwargs["api_base"] = config["base_url"]

    try:
        resp = await litellm.acompletion(**kwargs)
        text = resp.choices[0].message.content or ""
        parsed = _parse_json_reply(text)
    except (LlmCallError, json.JSONDecodeError):
        raise
    except Exception as e:  # litellm raises provider-specific exceptions
        raise LlmCallError(str(e))

    return compose_prd_markdown(parsed)
```

@router.post("/{issue_id}/prd/draft", status_code=202)
async def draft_issue_prd(
    issue_id: UUID,
    user: AuthUser = Depends(verify_token),
    settings: Settings = Depends(get_settings),
):
    """US-2.3: expand a feature's raw idea into a PRD artifact.

    The only PRD step that needs the API: it calls the org's thinking-LLM.
    Editing, approving, and sending back are plain RLS CRUD/RPC from the
    web app. A provider failure raises before save_prd_draft runs, so a
    failed draft never moves the issue.
    """
    issues = await postgrest_get(
        settings,
        user.token,
        "issues",
        {"select": "id,type,title,body,status", "id": f"eq.{issue_id}", "limit": "1"},
    )
    if not issues:
        raise HTTPException(status_code=404, detail="Issue not found")
    issue = issues[0]
    if issue["type"] != "feature":
        raise HTTPException(
            status_code=409,
            detail=f'only a feature issue has a PRD (this one is a {issue["type"]})',
        )

    # Informed retry (US-1.13's shape): the newest prd artifact and, if the
    # reviewer sent that exact draft back, their comment.
    previous_draft: str | None = None
    feedback: str | None = None
    artifacts = await postgrest_get(
        settings,
        user.token,
        "artifacts",
        {
            "select": "id,content",
            "issue_id": f"eq.{issue_id}",
            "kind": "eq.prd",
            "order": "version.desc",
            "limit": "1",
        },
    )
    if artifacts:
        previous_draft = artifacts[0]["content"]
        approvals = await postgrest_get(
            settings,
            user.token,
            "approvals",
            {
                "select": "comment",
                "gate": "eq.prd",
                "subject_type": "eq.artifact",
                "subject_id": f"eq.{artifacts[0]['id']}",
                "decision": "eq.sent-back",
                "order": "created_at.desc",
                "limit": "1",
            },
        )
        if approvals:
            feedback = approvals[0]["comment"]

    try:
        content = await draft_prd(
            settings,
            user.token,
            issue["title"],
            issue.get("body") or "",
            previous_draft=previous_draft if feedback else None,
            feedback=feedback,
        )
    except LlmNotConfigured:
        raise HTTPException(
            status_code=409,
            detail="No LLM provider configured — set one up in Settings.",
        )
    except LlmCallError as e:
        raise HTTPException(status_code=502, detail=f"LLM call failed: {e.message}")

    try:
        artifact_id = await rpc(
            settings,
            user.token,
            "save_prd_draft",
            {"p_issue": str(issue_id), "p_content": content},
        )
    except RpcError as e:
        if "not found" in e.message:
            raise HTTPException(status_code=404, detail="Issue not found")
        raise HTTPException(status_code=409, detail=e.message)

    return {"artifact_id": artifact_id, "content": content}
```

/** US-2.3: a PRD artifact is one markdown document with four fixed `##`
 * headings, composed by the API (see `compose_prd_markdown` in
 * apps/api/app/llm.py). These helpers split it for per-section editing and
 * put it back together byte-identically. The heading strings MUST stay in
 * sync with PRD_SECTIONS on the API side. */

export const PRD_SECTIONS = [
  { key: "problem", heading: "Problem" },
  { key: "goals", heading: "Goals" },
  { key: "out_of_scope", heading: "Out of scope" },
  { key: "acceptance_criteria", heading: "Acceptance criteria" },
] as const;

export type PrdSectionKey = (typeof PRD_SECTIONS)[number]["key"];
export type PrdSections = Record<PrdSectionKey, string>;

export const EMPTY_PRD_SECTIONS: PrdSections = {
  problem: "",
  goals: "",
  out_of_scope: "",
  acceptance_criteria: "",
};

/** Split a PRD document on its `## <heading>` lines. Anything under an
 * unrecognized heading (a provider or a PM inventing one) is kept by
 * appending it to the section it appeared under, so no text is ever lost. */
export function parsePrdSections(content: string): PrdSections {
  const sections: PrdSections = { ...EMPTY_PRD_SECTIONS };
  const lines = (content ?? "").split("\n");
  let current: PrdSectionKey | null = null;
  const buffers: Record<PrdSectionKey, string[]> = {
    problem: [],
    goals: [],
    out_of_scope: [],
    acceptance_criteria: [],
  };

  for (const line of lines) {
    const match = /^##\s+(.*?)\s*$/.exec(line);
    const known = match
      ? PRD_SECTIONS.find(
          (s) => s.heading.toLowerCase() === match[1].toLowerCase()
        )
      : undefined;
    if (known) {
      current = known.key;
      continue;
    }
    if (current) buffers[current].push(line);
  }

  for (const { key } of PRD_SECTIONS) {
    sections[key] = buffers[key].join("\n").trim();
  }
  return sections;
}

/** Inverse of parsePrdSections — and byte-identical to the API's
 * compose_prd_markdown for the same section values. */
export function serializePrdSections(sections: PrdSections): string {
  return PRD_SECTIONS.map(
    ({ key, heading }) => `## ${heading}\n\n${(sections[key] ?? "").trim()}`
  )
    .join("\n\n")
    .trim();
}

  issueId,

  issueStatus: string;
  artifacts: PrdArtifact[];
  approvals: PrdApproval[];

  const current = artifacts[0] ?? null;
  const older = artifacts.slice(1);

  const [editing, setEditing] = useState(false);

      router.refresh();

      // A failed draft leaves the issue exactly where it was (US-2.3).
      setError((e as Error).message);
    } finally {
      setBusy(null);
    }

    if (!current) return;
    setError(null);
    setBusy("save");

      const supabase = createClient();

        .from("artifacts")
        .update({ content: serializePrdSections(sections) })

      if (dbError) {

      setEditing(false);

      setBusy(null);

              {current ? "Re-draft PRD" : "Draft PRD"}

          </div>
        </div>
      </CardHeader>
      <CardContent className="grid gap-4">
        {lastSendBack && isDraft && (
          <div className="rounded-md border border-orange-200 bg-orange-50 p-3 text-sm dark:border-orange-900 dark:bg-orange-950/40">
            <p className="font-medium">
              Sent back {formatWhen(lastSendBack.created_at)}
            </p>
            <p className="text-muted-foreground">{lastSendBack.comment}</p>
          </div>
        )}

        {!current && (
          <EmptyState
            icon={FileText}
            title="No PRD yet"
            description="Draft one from this feature's description, then edit and approve it."
          />
        )}

        {current && editing && (
          <div className="grid gap-4">
            {PRD_SECTIONS.map(({ key, heading }) => (
              <div key={key} className="grid gap-2">
                <Label htmlFor={`prd-${key}`}>{heading}</Label>
                <Textarea
                  id={`prd-${key}`}
                  rows={key === "problem" ? 5 : 4}
                  value={sections[key]}
                  onChange={(e) =>
                    setSections((prev) => ({ ...prev, [key]: e.target.value }))
                  }
                />
              </div>
            ))}
            <div className="flex items-center gap-2">
              <Button onClick={handleSave} disabled={busy !== null} size="sm">
                {busy === "save" && <Loader2 className="size-4 animate-spin" />}
                Save draft
              </Button>
              <Button
                variant="ghost"
                size="sm"
                onClick={() => {
                  setSections(parsePrdSections(current.content));
                  setEditing(false);
                }}
              >
                Cancel
              </Button>
            </div>
          </div>
        )}

        {current && !editing && (
          <div className={PROSE}>
            <ReactMarkdown>{current.content}</ReactMarkdown>
          </div>
        )}

        {current && !editing && isDraft && (
          <div className="grid gap-2 border-t pt-4">
            <div className="flex items-center gap-2">
              <Button onClick={handleApprove} disabled={busy !== null} size="sm">
                {busy === "approve" ? (
                  <Loader2 className="size-4 animate-spin" />
                ) : (
                  <Check className="size-4" />
                )}
                Approve PRD
              </Button>
              <Button
                variant="outline"
                size="sm"
                onClick={() => setSendBackOpen((v) => !v)}
                disabled={busy !== null}
              >
                <Undo2 className="size-4" />

    </Card>
  );
}
